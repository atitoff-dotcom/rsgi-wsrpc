from rsgi_wsrpc.plugins.smart_cache import invalidates

@app.rpc("tasks.toggle")
@invalidates(tags=["tasks", "task.{id}"])

    clientCache.delete('tasks.list');
    reloadTasksUI();
  }
});

🌊 5. Multi-return Streaming (stream: true)

Push intermediate progress chunks inside a single JSON-RPC call without blocking the WebSocket pipeline.

stream: true • Non-blocking
Unlike HTTP REST requiring long-polling or SSE connections, handlers stream chunks via session.send_stream_chunk within the active rpc_id.
Status: Idle 0%
--:--:--Intermediate stream chunk log...
@app.rpc("stream.heavy_job")
async def heavy_job(session: JsonRpcSession):
    rpc_id = current_rpc_id_ctx.get()
    for pct in [25, 50, 75]:
        await asyncio.sleep(0.4)
        # Push intermediate chunk to client:
        await session.send_stream_chunk(rpc_id, {"progress": pct})

    # Final return resolves the overall RPC call:
    return {"progress": 100, "status": "done"}
// Client registers chunk callback:
await rpc.requestStream('stream.heavy_job', {}, (chunk) => {
  console.log('Intermediate progress:', chunk.progress);
  updateProgressBar(chunk.progress);
});
// Chunk 1 (intermediate):
{"jsonrpc":"2.0", "result":{"stream":true, "data":{"progress":25}}, "id":4}

// Chunk 2 (intermediate):
{"jsonrpc":"2.0", "result":{"stream":true, "data":{"progress":50}}, "id":4}

// Final result:
{"jsonrpc":"2.0", "result":{"progress":100, "status":"done"}, "id":4}

🪞 6. Symmetric Reverse RPC (Server → Client)

Full-duplex JSON-RPC 2.0: the server initiates calls to the browser via session.send_request() and awaits responses.

Symmetric WSRPC • Bidirectional
No bespoke protocols required: the server queries browser viewport dimensions, timezone, or telemetry and awaits the answer seamlessly.
--:--:--Click button to trigger server-to-client call...
@app.rpc("reverse_rpc.ask_client")
async def ask_client(session: JsonRpcSession):
    # Server initiates request to browser and awaits result!
    client_res = await session.send_request(
        method="client.get_env",
        params={"server_now": time.time()},
        timeout=4.0
    )
    return {"received_from_browser": client_res}
// Client method registration:
rpc.registerMethod('client.get_env', (params) => {
  return {
    userAgent: navigator.userAgent,
    screen: `${window.innerWidth}x${window.innerHeight}`,
    browserTime: new Date().toISOString()
  };
});
// 1. Server sends request to client (generated id):
<-- {"jsonrpc":"2.0", "method":"client.get_env", "params":{...}, "id": 1001}

// 2. Browser responds to server:
--> {"jsonrpc":"2.0", "result":{"screen":"1920x1080"}, "id": 1001}

@http_route("/upload", ["POST"])
async def upload_handler(scope, proto):
    # Streaming chunks directly to disk file (O(1) RAM):
    bytes_count, sha256 = await stream_request_to_disk(proto, dest_path)
    proto.response_str(200, [("content-type", "application/json")],
                       orjson.dumps({"bytes": bytes_count, "sha256": sha256}))
// Stream file body via Fetch API:

  method: 'POST',

const data = await resp.json();
